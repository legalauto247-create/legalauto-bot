function doGet(e) {
  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheets()[0];

  const rows = sheet
    .getDataRange()
    .getValues();

  const headers = rows.shift();

  let products = rows.map(row => {
    let obj = {};

    headers.forEach((h, i) => {
      obj[h] = row[i];
    });

    return obj;
  });

  products = products.filter(p => {
    return String(p.miniapp_visible).toUpperCase() !== "FALSE";
  });

  if (e.parameter.q) {
    const q = String(e.parameter.q).toLowerCase();

    products = products.filter(p =>
      String(p.search_text || JSON.stringify(p))
        .toLowerCase()
        .includes(q)
    );
  }

  if (e.parameter.brand) {
    products = products.filter(p =>
      String(p.brand).toLowerCase() === String(e.parameter.brand).toLowerCase()
    );
  }

  if (e.parameter.body) {
    products = products.filter(p =>
      String(p.body).toLowerCase().includes(String(e.parameter.body).toLowerCase())
    );
  }

  if (e.parameter.category) {
    products = products.filter(p =>
      String(p.category).toLowerCase() === String(e.parameter.category).toLowerCase()
    );
  }

  const brands = [...new Set(products.map(p => p.brand).filter(Boolean))];
  const bodies = [...new Set(products.map(p => p.body).filter(Boolean))];
  const categories = [...new Set(products.map(p => p.category).filter(Boolean))];
  const conditions = [...new Set(products.map(p => p.condition).filter(Boolean))];

  return ContentService
    .createTextOutput(JSON.stringify({
      products,
      filters: {
        brands,
        bodies,
        categories,
        conditions
      }
    }))
    .setMimeType(ContentService.MimeType.JSON);
}

function migrateToV6() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];

  const headers = sheet.getRange(1,1,1,sheet.getLastColumn()).getValues()[0];

  const add = [
    "supplier","series","display_car","compatibility","condition","in_stock","stock_location",
    "photo_cover","photo_1","photo_2","photo_3","photo_4","photo_5",
    "search_text","priority","featured","miniapp_visible",
    "published_at","telegram_post","telegram_message_id","post_hash",
    "views","leads","favorite","source","sort_order"
  ];

  let col = headers.length;

  add.forEach(name => {
    if (!headers.includes(name)) {
      col++;
      sheet.getRange(1,col).setValue(name);
    }
  });

  const data = sheet.getDataRange().getValues();
  const h = data[0];
  const ix = name => h.indexOf(name);

  function detectCar(modelRaw, bodyRaw, textRaw) {
    const model = String(modelRaw || "").trim().toUpperCase();
    const body = String(bodyRaw || "").trim().toUpperCase();
    const text = String(textRaw || "").toUpperCase();

    const knownBodies = ["G05","G06","G07","G30","G11","F15","F16"];
    const bodyToSeries = {
      G05:"X5",
      G06:"X6",
      G07:"X7",
      G30:"5 Series",
      G11:"7 Series",
      F15:"X5",
      F16:"X6"
    };

    let detectedBody = "";

    for (const b of knownBodies) {
      if (model === b || body === b || text.includes(b)) {
        detectedBody = b;
        break;
      }
    }

    let series = "";

    if (detectedBody) {
      series = bodyToSeries[detectedBody] || "";
    }

    if (!series && model.startsWith("X")) {
      series = model;
    }

    if (!detectedBody && body.startsWith("G")) {
      detectedBody = body;
    }

    return {
      series,
      body: detectedBody || body || model
    };
  }

  for (let r = 1; r < data.length; r++) {
    const brand = data[r][ix("brand")] || "";
    const model = data[r][ix("model")] || "";
    const bodyOld = data[r][ix("body")] || "";
    const name = data[r][ix("name")] || "";
    const description = data[r][ix("description")] || "";
    const oem = data[r][ix("oem")] || "";
    const qty = Number(data[r][ix("qty")] || 0);
    const photo = data[r][ix("photo")] || "";

    const detected = detectCar(model, bodyOld, `${name} ${description}`);

    sheet.getRange(r+1, ix("series")+1).setValue(detected.series);
    sheet.getRange(r+1, ix("body")+1).setValue(detected.body);

    sheet.getRange(r+1, ix("display_car")+1)
      .setValue(`${brand} ${detected.series} ${detected.body}`.trim());

    sheet.getRange(r+1, ix("supplier")+1).setValue("Own");
    sheet.getRange(r+1, ix("condition")+1).setValue("Оригинал Б/У");
    sheet.getRange(r+1, ix("in_stock")+1).setValue(qty > 0);
    sheet.getRange(r+1, ix("stock_location")+1).setValue("Москва");
    sheet.getRange(r+1, ix("photo_cover")+1).setValue(photo);
    sheet.getRange(r+1, ix("photo_1")+1).setValue(photo);

    sheet.getRange(r+1, ix("search_text")+1)
      .setValue(`${brand} ${detected.series} ${detected.body} ${name} ${oem}`);

    sheet.getRange(r+1, ix("priority")+1).setValue("NORMAL");
    sheet.getRange(r+1, ix("featured")+1).setValue("FALSE");
    sheet.getRange(r+1, ix("miniapp_visible")+1).setValue(true);
    sheet.getRange(r+1, ix("post_hash")+1).setValue(`${brand}|${oem}`);
    sheet.getRange(r+1, ix("views")+1).setValue(0);
    sheet.getRange(r+1, ix("leads")+1).setValue(0);
    sheet.getRange(r+1, ix("favorite")+1).setValue(0);
    sheet.getRange(r+1, ix("source")+1).setValue("Telegram");
    sheet.getRange(r+1, ix("sort_order")+1).setValue(0);
  }

  SpreadsheetApp.getUi().alert("V6 обновлена корректно");
}

function cleanCarDisplay() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  const data = sheet.getDataRange().getValues();
  const h = data[0];
  const ix = name => h.indexOf(name);

  const bodyToSeries = {
    G01: "X3",
    G05: "X5",
    G06: "X6",
    G07: "X7",
    G30: "5 Series",
    G11: "7 Series",
    F15: "X5",
    F16: "X6"
  };

  const knownBodies = Object.keys(bodyToSeries);

  for (let r = 1; r < data.length; r++) {
    const brand = String(data[r][ix("brand")] || "").trim();
    const model = String(data[r][ix("model")] || "").trim().toUpperCase();
    const bodyOld = String(data[r][ix("body")] || "").trim().toUpperCase();
    const name = String(data[r][ix("name")] || "").toUpperCase();
    const desc = String(data[r][ix("description")] || "").toUpperCase();

    const text = `${model} ${bodyOld} ${name} ${desc}`;

    let body = "";
    let series = "";

    for (const b of knownBodies) {
      if (text.includes(b)) {
        body = b;
        series = bodyToSeries[b];
        break;
      }
    }

    if (!series) {
      if (["X3","X5","X6","X7"].includes(model)) series = model;
      else if (["X3","X5","X6","X7"].includes(bodyOld)) series = bodyOld;
      else if (model) series = model;
      else if (bodyOld) series = bodyOld;
    }

    let display = "";

    if (body && series) {
      display = `${brand} ${series} ${body}`.trim();
    } else if (series) {
      display = `${brand} ${series}`.trim();
    } else {
      display = brand;
    }

    sheet.getRange(r + 1, ix("series") + 1).setValue(series);
    sheet.getRange(r + 1, ix("body") + 1).setValue(body || "");
    sheet.getRange(r + 1, ix("display_car") + 1).setValue(display);

    sheet.getRange(r + 1, ix("search_text") + 1).setValue(
      `${brand} ${series} ${body} ${data[r][ix("name")] || ""} ${data[r][ix("oem")] || ""}`
    );
  }

  SpreadsheetApp.getUi().alert("Авто очищены корректно");
}

function fixCompatibility() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  const data = sheet.getDataRange().getValues();
  const h = data[0];
  const ix = name => h.indexOf(name);

  const modelToBody = {
    "X3": "G01",
    "X5": "G05",
    "X6": "G06",
    "X7": "G07",
    "5 SERIES": "G30"
  };

  for (let r = 1; r < data.length; r++) {
    const brand = String(data[r][ix("brand")] || "").trim();
    const name = String(data[r][ix("name")] || "");
    const desc = String(data[r][ix("description")] || "");
    const oem = String(data[r][ix("oem")] || "");

    const text = `${name} ${desc}`.toUpperCase();

    let foundSeries = [];
    let foundBodies = [];

    Object.keys(modelToBody).forEach(model => {
      if (text.includes(model)) {
        foundSeries.push(model === "5 SERIES" ? "5 Series" : model);
        foundBodies.push(modelToBody[model]);
      }
    });

    ["G01","G05","G06","G07","G30"].forEach(body => {
      if (text.includes(body) && !foundBodies.includes(body)) {
        foundBodies.push(body);
      }
    });

    if (foundSeries.length === 0) {
      const oldSeries = String(data[r][ix("series")] || "").trim();
      if (oldSeries) foundSeries.push(oldSeries);
    }

    if (foundBodies.length === 0) {
      const oldBody = String(data[r][ix("body")] || "").trim();
      if (oldBody && oldBody !== foundSeries[0]) foundBodies.push(oldBody);
    }

    const series = [...new Set(foundSeries)].join("|");
    const bodies = [...new Set(foundBodies)].join("|");

    const displayCar = series
      ? `${brand} ${series.replaceAll("|", " • ")}`
      : brand;

    const compatibility = [...new Set(foundSeries)].map(s => {
      const b = modelToBody[String(s).toUpperCase()] || "";
      return `${brand} ${s}${b ? " " + b : ""}`;
    }).join("; ");

    const searchText = [
      brand,
      series.replaceAll("|", " "),
      bodies.replaceAll("|", " "),
      compatibility,
      name,
      oem
    ].join(" ");

    sheet.getRange(r + 1, ix("series") + 1).setValue(series);
    sheet.getRange(r + 1, ix("body") + 1).setValue(bodies);
    sheet.getRange(r + 1, ix("display_car") + 1).setValue(displayCar);
    sheet.getRange(r + 1, ix("compatibility") + 1).setValue(compatibility);
    sheet.getRange(r + 1, ix("search_text") + 1).setValue(searchText);
  }

  SpreadsheetApp.getUi().alert("Совместимость исправлена");
}

function importPhotosFromDrive() {

  const ROOT_FOLDER_ID = "1jJ9Dkv2tvJJmwomEm6_IpSeD3-Xwarez";
  const MAX_PHOTOS_PER_PART = 5;
  const LOG_EVERY_FOLDERS = 20;
  const LOG_EVERY_UPDATED_ROWS = 10;

  const sheet = SpreadsheetApp
    .getActiveSpreadsheet()
    .getSheets()[0];

  const rows = sheet
    .getDataRange()
    .getValues();

  const headers = rows[0];

  const ix = name => headers.indexOf(name);

  const requiredColumns = [
    "oem",
    "photo_cover",
    "photo_1",
    "photo_2",
    "photo_3",
    "photo_4",
    "photo_5"
  ];

  requiredColumns.forEach(col => {
    if (ix(col) === -1) {
      throw new Error("Нет колонки в таблице: " + col);
    }
  });

  const stats = {
    foldersChecked: 0,
    matchedFolders: 0,
    rowsUpdated: 0,
    photosAdded: 0,
    skippedNoPhotos: 0,
    skippedNoOemInSheet: 0,
    duplicateOemFolders: 0,
    startedAt: new Date()
  };

  const oemToRow = {};
  const importedOems = {};

  for (let r = 1; r < rows.length; r++) {
    const oem = normalizeOem(rows[r][ix("oem")]);

    if (oem) {
      oemToRow[oem] = r + 1;
    }
  }

  logProgress("СТАРТ ИМПОРТА");

  const root = DriveApp.getFolderById(ROOT_FOLDER_ID);

  scanFolder(root);

  SpreadsheetApp.flush();

  logProgress("ИМПОРТ ЗАВЕРШЕН");

  return stats;

  function scanFolder(folder) {
    const folders = folder.getFolders();

    while (folders.hasNext()) {
      const currentFolder = folders.next();

      stats.foldersChecked++;

      const folderName = currentFolder.getName();
      const oem = extractOemFromFolderName(folderName);

      if (stats.foldersChecked % LOG_EVERY_FOLDERS === 0) {
        logProgress("Проверяю папки Drive...");
      }

      if (oem) {
        if (oemToRow[oem]) {

          if (importedOems[oem]) {
            stats.duplicateOemFolders++;
            console.log(
              "⚠️ Дубль OEM в Drive: " +
              oem +
              " | папка: " +
              folderName
            );
          }

          importedOems[oem] = true;

          stats.matchedFolders++;

          console.log(
            "✅ Найден OEM: " +
            oem +
            " | строка: " +
            oemToRow[oem] +
            " | папка: " +
            folderName
          );

          importPhotosFromPartFolder(
            currentFolder,
            oemToRow[oem],
            oem
          );

        } else {
          stats.skippedNoOemInSheet++;

          console.log(
            "⏭ OEM есть в Drive, но нет в таблице: " +
            oem +
            " | папка: " +
            folderName
          );
        }
      }

      scanFolder(currentFolder);
    }
  }

  function importPhotosFromPartFolder(folder, rowNumber, oem) {
    const files = folder.getFiles();

    const photos = [];

    while (files.hasNext()) {
      const file = files.next();

      const mime = file.getMimeType();

      if (!mime || !mime.startsWith("image/")) {
        continue;
      }

      const fileId = file.getId();

      photos.push({
        name: file.getName(),
        url:
          "https://drive.google.com/thumbnail?id=" +
          fileId +
          "&sz=w1200"
      });
    }

    photos.sort((a, b) =>
      naturalSort(a.name, b.name)
    );

    const selectedPhotos =
      photos.slice(0, MAX_PHOTOS_PER_PART);

    if (selectedPhotos.length === 0) {
      stats.skippedNoPhotos++;

      console.log(
        "⚠️ Фото не найдены: " +
        oem +
        " | строка: " +
        rowNumber
      );

      return;
    }

    sheet
      .getRange(rowNumber, ix("photo_cover") + 1)
      .setValue(selectedPhotos[0].url);

    for (let i = 0; i < MAX_PHOTOS_PER_PART; i++) {
      const colName = "photo_" + (i + 1);
      const url = selectedPhotos[i]
        ? selectedPhotos[i].url
        : "";

      sheet
        .getRange(rowNumber, ix(colName) + 1)
        .setValue(url);
    }

    stats.rowsUpdated++;
    stats.photosAdded += selectedPhotos.length;

    console.log(
      "📸 Записано фото: " +
      selectedPhotos.length +
      " | OEM: " +
      oem +
      " | строка: " +
      rowNumber
    );

    if (stats.rowsUpdated % LOG_EVERY_UPDATED_ROWS === 0) {
      SpreadsheetApp.flush();
      logProgress("Промежуточный прогресс");
    }
  }

  function extractOemFromFolderName(name) {
    const normalized = String(name || "")
      .trim()
      .toUpperCase();

    const match = normalized.match(/[A-Z0-9]{7,}/);

    return match ? match[0] : "";
  }

  function normalizeOem(value) {
    return String(value || "")
      .trim()
      .toUpperCase()
      .replace(/\s+/g, "");
  }

  function naturalSort(a, b) {
    return String(a).localeCompare(
      String(b),
      "ru",
      {
        numeric: true,
        sensitivity: "base"
      }
    );
  }

  function logProgress(stage) {
    const seconds = Math.round(
      (new Date() - stats.startedAt) / 1000
    );

    console.log(
      "\n" +
      "==============================\n" +
      stage + "\n" +
      "Папок проверено: " + stats.foldersChecked + "\n" +
      "OEM найдено в таблице: " + stats.matchedFolders + "\n" +
      "Строк обновлено: " + stats.rowsUpdated + "\n" +
      "Фото записано: " + stats.photosAdded + "\n" +
      "Без фото: " + stats.skippedNoPhotos + "\n" +
      "OEM нет в таблице: " + stats.skippedNoOemInSheet + "\n" +
      "Дубли OEM в Drive: " + stats.duplicateOemFolders + "\n" +
      "Время: " + seconds + " сек.\n" +
      "=============================="
    );
  }
}

  function chooseBestRows(rows, folderPath) {
    if (rows.length === 1) return rows;

    const path = String(folderPath || "").toUpperCase();

    const scored = rows.map(rowNumber => {
      const row = sheet.getRange(rowNumber, 1, 1, sheet.getLastColumn()).getValues()[0];

      let score = 0;

      const brand = String(row[ix("brand")] || "").toUpperCase();
      const series = String(row[ix("series")] || "").toUpperCase();
      const body = String(row[ix("body")] || "").toUpperCase();

      if (brand && path.includes(brand)) score += 2;

      splitTokens(series).forEach(x => {
        if (path.includes(x)) score += 4;
      });

      splitTokens(body).forEach(x => {
        if (path.includes(x)) score += 5;
      });

      return { rowNumber, score };
    });

    scored.sort((a, b) => b.score - a.score);

    if (scored[0].score === 0) return [];

    return scored
      .filter(x => x.score === scored[0].score)
      .map(x => x.rowNumber);
  }

  function getPhotoLinks(folder) {
    const files = folder.getFiles();
    const photos = [];

    while (files.hasNext()) {
      const file = files.next();
      const mime = file.getMimeType();

      if (!mime || !mime.startsWith("image/")) continue;

      photos.push({
        name: file.getName(),
        url: "https://drive.google.com/thumbnail?id=" + file.getId() + "&sz=w1200"
      });
    }

    photos.sort((a, b) =>
      String(a.name).localeCompare(String(b.name), "ru", {
        numeric: true,
        sensitivity: "base"
      })
    );

    return photos;
  }

  function parsePartFolder(folderName, path) {
    const normalized = String(folderName || "").trim();
    const upper = normalized.toUpperCase();

    const match = upper.match(/[A-Z0-9]{7,}/);
    const oem = match ? match[0] : "";

    const name = oem
      ? normalized.replace(new RegExp(oem, "i"), "").trim()
      : normalized;

    return {
      oem,
      name,
      brand: detectBrand(path),
      series: detectSeries(path),
      body: detectBody(path)
    };
  }

  function detectBrand(text) {
    const t = String(text || "").toUpperCase();

    if (t.includes("BMW")) return "BMW";
    if (t.includes("GEELY")) return "Geely";
    if (t.includes("LI AUTO") || t.includes("LIXIANG")) return "Li Auto";
    if (t.includes("MERCEDES")) return "Mercedes";
    if (t.includes("AUDI")) return "Audi";

    return "";
  }

  function detectSeries(text) {
    const t = String(text || "").toUpperCase();
    const map = ["X3", "X5", "X6", "X7", "5 SERIES", "7 SERIES", "COOLRAY", "MONJARO", "TUGELLA", "ATLAS", "L6", "L7", "L8", "L9"];

    for (const m of map) {
      if (t.includes(m)) return m;
    }

    return "";
  }

  function detectBody(text) {
    const t = String(text || "").toUpperCase();
    const match = t.match(/\b(G01|G05|G06|G07|G30|G11|F15|F16|F95|F96)\b/);
    return match ? match[1] : "";
  }

  function splitTokens(value) {
    return String(value || "")
      .toUpperCase()
      .split("|")
      .map(x => x.trim())
      .filter(Boolean);
  }

  function normalize(value) {
    return String(value || "")
      .trim()
      .toUpperCase()
      .replace(/\s+/g, "");
  }

  function setCell(rowNumber, colName, value) {
    const col = ix(colName);
    if (col >= 0) {
      sheet.getRange(rowNumber, col + 1).setValue(value);
    }
  }

  function ensureColumns(sheet, columns) {
    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

    let col = headers.length;

    columns.forEach(name => {
      if (!headers.includes(name)) {
        col++;
        sheet.getRange(1, col).setValue(name);
      }
    });
  }

  function appendRows(ss, sheetName, headers, rows) {
    let sh = ss.getSheetByName(sheetName);

    if (!sh) {
      sh = ss.insertSheet(sheetName);
      sh.appendRow(headers);
    }

    if (rows.length) {
      sh.getRange(sh.getLastRow() + 1, 1, rows.length, headers.length).setValues(rows);
    }
  }

function translateSheet(ss, sourceName, targetName, dict) {
  const source = ss.getSheetByName(sourceName);
  if (!source) return;

  let target = ss.getSheetByName(targetName);
  if (!target) target = ss.insertSheet(targetName);

  target.clear();

  const values = source.getDataRange().getValues();
  if (!values.length) return;

  const headers = values[0].map(h => dict[h] || h);
  const output = [headers, ...values.slice(1)];

  target
    .getRange(1, 1, output.length, output[0].length)
    .setValues(output);

  target.autoResizeColumns(1, output[0].length);
}

  const folderId = extractDriveId(folderUrl);
  if (!folderId) return [];

  const folder = DriveApp.getFolderById(folderId);
  const files = folder.getFiles();
  const photos = [];

  while (files.hasNext()) {
    const file = files.next();
    const mime = file.getMimeType();

    photos.push({
      name: file.getName(),

function getNextId(sheet, ix) {
  const data = sheet.getDataRange().getValues();
  const idCol = ix("id");

  let maxId = 0;

  if (idCol === -1) return "";

  for (let r = 1; r < data.length; r++) {

    if (id > maxId) maxId = id;
  }

  return maxId + 1;
}

function calculateScore(oem, name, price, photoCount, brand, series, body) {
  let score = 0;

  if (photoCount > 0) score += 20;
  if (Number(price || 0) > 0) score += 20;
  if (oem) score += 20;
  if (name && String(name).length > 2) score += 20;

function guessCategory(name) {
  const t = String(name || "").toLowerCase();

  if (t.includes("бампер") || t.includes("двер") || t.includes("крыл") || t.includes("капот") || t.includes("молдинг")) {
    return "Кузов";
  }

  if (t.includes("фара") || t.includes("фонар") || t.includes("птф")) {
    return "Оптика";
  }

  if (t.includes("провод") || t.includes("блок") || t.includes("камера") || t.includes("датчик")) {
    return "Электрика";
  }

  if (t.includes("рычаг") || t.includes("стойк") || t.includes("пневмо") || t.includes("амортиз")) {
    return "Подвеска";
  }

  return "Прочее";
}

  const existing = {};

  const sourceSeen = {};

  const report = [[

    "OEM",

    "Название",

    "Причина",

    "Строка в основном листе",

    "Строка в NEW_PARTS_FROM_DRIVE",

    "Путь папки"

  ]];

  for (let r = 1; r < mainData.length; r++) {

    const key = buildKey(

      mainData[r][mix("oem")],

      mainData[r][mix("name")]

    );

    if (!existing[key]) existing[key] = [];

    existing[key].push(r + 1);

  }

  for (let r = 1; r < sourceData.length; r++) {

    const oem = clean(sourceData[r][six("oem")]);

    const name = clean(sourceData[r][six("name_from_folder")]);

    const folder = clean(sourceData[r][six("folder")]);

    const key = buildKey(oem, name);

    if (sourceSeen[key]) {

      report.push([

        oem,

        name,

        "Дубль внутри NEW_PARTS_FROM_DRIVE",

        "",

        r + 1,

        folder

      ]);

      continue;

    }

    sourceSeen[key] = true;

    if (existing[key]) {

      report.push([

        oem,

        name,

        "Уже есть в основном листе",

        existing[key].join(", "),

        r + 1,

        folder

      ]);

    }

  }

  let sh = ss.getSheetByName("АНАЛИЗ_ДУБЛЕЙ_DRIVE");

  if (!sh) sh = ss.insertSheet("АНАЛИЗ_ДУБЛЕЙ_DRIVE");

  sh.clear();

  sh.getRange(1, 1, report.length, report[0].length).setValues(report);

  sh.autoResizeColumns(1, report[0].length);

  console.log("Готово. Найдено дублей/повторов: " + (report.length - 1));

}
